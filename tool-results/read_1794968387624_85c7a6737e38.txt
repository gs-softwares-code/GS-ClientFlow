    79→function birthdayInputValue(c: { birthdayYear?: number | null; birthdayMonth: number | null; birthdayDay: number | null }): string {
    80→  if (!c.birthdayMonth || !c.birthdayDay) return "";
    81→  const year = c.birthdayYear ?? new Date().getFullYear();
    82→  return `${year}-${String(c.birthdayMonth).padStart(2, "0")}-${String(c.birthdayDay).padStart(2, "0")}`;
    83→}
    84→
    85→function ruleValueText(value: string | number | (string | number)[]): string {
    86→  return Array.isArray(value) ? value.join(",") : String(value);
    87→}
    88→
    89→function toneBorder(tone?: string | null): string {
    90→  if (tone === "positive") return "border-emerald-500";
    91→  if (tone === "negative") return "border-rose-500";
    92→  return "border-border";
    93→}
    94→
    95→function messageStatusClass(status: string): string {
    96→  switch (status) {
    97→    case "delivered": case "read": return "border-emerald-600 text-emerald-700";
    98→    case "failed": return "border-rose-600 text-rose-700";
    99→    case "queued": case "sent": return "border-amber-600 text-amber-700";
   100→    default: return "";
   101→  }
   102→}

   124→export default function CustomersView({ ctx }: { ctx: AppCtx }) {
   125→  const { toast } = useToast();
   126→
   127→  const [searchInput, setSearchInput] = useState("");
   128→  const [search, setSearch] = useState("");
   129→  const [segmentId, setSegmentId] = useState("");
   130→  const [source, setSource] = useState("");
   131→  const [status, setStatus] = useState("active");
   132→  const [birthdayChip, setBirthdayChip] = useState(false);
   133→  const [inactiveChip, setInactiveChip] = useState(false);
   134→
   135→  const [items, setItems] = useState<CustomerListItem[]>([]);
   136→  const [nextCursor, setNextCursor] = useState<string | null>(null);
   137→  const [hasMore, setHasMore] = useState(false);
   138→  const [loading, setLoading] = useState(true);
   139→  const [loadingMore, setLoadingMore] = useState(false);
   140→  const [listError, setListError] = useState<string | null>(null);
   141→  const requestSeq = useRef(0);
   142→
   143→  const [segments, setSegments] = useState<SegmentDto[]>([]);
   144→  const [segmentsLoading, setSegmentsLoading] = useState(true);
   145→  const [detailId, setDetailId] = useState<string | null>(null);
   146→  const [addOpen, setAddOpen] = useState(false);
   147→  const [segmentOpen, setSegmentOpen] = useState(false);
   148→
   149→  // 400ms debounce on the search box
   150→  useEffect(() => {
   151→    const t = setTimeout(() => setSearch(searchInput.trim()), 400);
   152→    return () => clearTimeout(t);
   153→  }, [searchInput]);
   154→
   155→  const filterParams = useCallback((cursor?: string | null) => {
   156→    const p = new URLSearchParams({ limit: String(PAGE_SIZE) });
   157→    if (cursor) p.set("cursor", cursor);
   158→    if (search) p.set("search", search);
   159→    if (segmentId) p.set("segmentId", segmentId);
   160→    if (source) p.set("source", source);
   161→    if (status !== "all") p.set("status", status);
   162→    if (birthdayChip) p.set("birthdayWithinDays", "14");
   163→    if (inactiveChip) p.set("inactiveWithinDays", "45");
   164→    return p;
   165→  }, [search, segmentId, source, status, birthdayChip, inactiveChip]);
   166→
   167→  const fetchPage = useCallback(async (cursor: string | null, mode: "replace" | "append" | "silent") => {
   168→    const seq = ++requestSeq.current;
   169→    if (mode === "replace") setLoading(true);
   170→    if (mode === "append") setLoadingMore(true);
   171→    try {
   172→      const res = await apiGet<CustomerListResponse>(`/api/customers?${filterParams(cursor).toString()}`);
   173→      if (seq !== requestSeq.current) return;
   174→      setItems((prev) => (mode === "append" ? [...prev, ...res.items] : res.items));
   175→      setNextCursor(res.nextCursor);
   176→      setHasMore(res.hasMore);
   177→      setListError(null);
   178→    } catch (err) {
   179→      if (seq !== requestSeq.current) return;
   180→      const message = (err as Error).message;
   181→      if (mode !== "silent") {
   182→        setListError(message);
   183→        toast({ title: "Could not load customers", description: message, variant: "destructive" });
   184→      }
   185→    } finally {
   186→      if (seq === requestSeq.current) { setLoading(false); setLoadingMore(false); }
   187→    }
   188→  }, [filterParams, toast]);
   189→
   190→  // any filter change refetches from scratch (cursor reset)
   191→  useEffect(() => { void fetchPage(null, "replace"); }, [fetchPage]);
   192→
   193→  const loadMore = () => { if (nextCursor && !loadingMore) void fetchPage(nextCursor, "append"); };
   194→  const refreshSilently = useCallback(() => { void fetchPage(null, "silent"); }, [fetchPage]);
   195→
   196→  const loadSegments = useCallback(async () => {
   197→    setSegmentsLoading(true);
   198→    try {
   199→      const res = await apiGet<{ segments: SegmentDto[] }>("/api/segments");
   200→      setSegments(res.segments);
   201→    } catch (err) {
   202→      toast({ title: "Could not load segments", description: (err as Error).message, variant: "destructive" });
   203→    } finally {
   204→      setSegmentsLoading(false);
   205→    }
   206→  }, [toast]);
   207→
   208→  useEffect(() => { void loadSegments(); }, [loadSegments]);
   209→
   210→  const deleteSegment = async (seg: SegmentDto) => {
   211→    try {
   212→      await apiDelete(`/api/segments?id=${encodeURIComponent(seg.id)}`);
   213→      toast({ title: "Segment deleted", description: `${seg.name} was removed from the rules engine` });
   214→      if (segmentId === seg.id) setSegmentId("");
   215→      void loadSegments();
   216→    } catch (err) {
   217→      toast({ title: "Could not delete segment", description: (err as Error).message, variant: "destructive" });
   218→    }
   219→  };
   220→
   221→  return (
   222→    <div className="space-y-4">
   223→      {/* Header row */}
   224→      <div className="flex flex-wrap items-center gap-3">
   225→        <div className="flex items-center gap-2.5">
   226→          <h1 className="text-2xl font-semibold tracking-tight">Customers</h1>
   227→          <Badge variant="secondary" className="tabular-nums">{loading ? "…" : `${items.length}${hasMore ? "+" : ""}`}</Badge>
   228→        </div>
   229→        <div className="ml-auto flex flex-wrap items-center gap-2">
   230→          <Button variant="outline" className="h-10 gap-1.5" onClick={() => downloadFile("/api/export/customers", `customers-${ctx.tenant.slug}.csv`)}>
   231→            <Download className="h-4 w-4" /> Export CSV
   232→          </Button>
   233→          <Button variant="outline" className="h-10 gap-1.5" onClick={() => setSegmentOpen(true)}>
   234→            <TagsIcon className="h-4 w-4" /> New segment
   235→          </Button>
   236→          <Button className="h-10 gap-1.5 bg-[var(--brand)] text-white hover:opacity-90" onClick={() => setAddOpen(true)}>
   237→            <Plus className="h-4 w-4" /> Add customer
   238→          </Button>
   239→        </div>
   240→      </div>
   241→
   242→      {/* Filter bar */}
   243→      <div className="flex flex-col gap-2 rounded-xl border bg-card p-3 sm:flex-row sm:flex-wrap sm:items-center">
   244→        <div className="relative w-full sm:max-w-xs">
   245→          <Search className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   246→          <Input aria-label="Search customers by name, email or phone" className="h-10 pl-9" placeholder="Search name, email or phone" value={searchInput} onChange={(e) => setSearchInput(e.target.value)} />
   247→        </div>
   248→        <Select value={segmentId || "all"} onValueChange={(v) => setSegmentId(v === "all" ? "" : v)}>
   249→          <SelectTrigger aria-label="Filter by segment" className="h-10 w-full sm:w-[180px]"><SelectValue /></SelectTrigger>
   250→          <SelectContent>
   251→            <SelectItem value="all">All segments</SelectItem>
   252→            {segments.map((s) => <SelectItem key={s.id} value={s.id}>{s.name} ({s.memberCount})</SelectItem>)}
   253→          </SelectContent>
   254→        </Select>
   255→        <Select value={source || "all"} onValueChange={(v) => setSource(v === "all" ? "" : v)}>
   256→          <SelectTrigger aria-label="Filter by source" className="h-10 w-full sm:w-[160px]"><SelectValue /></SelectTrigger>
   257→          <SelectContent>
   258→            <SelectItem value="all">All sources</SelectItem>
   259→            {SOURCES.map((s) => <SelectItem key={s} value={s}>{sourceLabel(s)}</SelectItem>)}
   260→          </SelectContent>
   261→        </Select>
   262→        <Select value={status} onValueChange={setStatus}>
   263→          <SelectTrigger aria-label="Filter by status" className="h-10 w-full sm:w-[150px]"><SelectValue /></SelectTrigger>
   264→          <SelectContent>
   265→            <SelectItem value="active">Active only</SelectItem>
   266→            <SelectItem value="archived">Archived only</SelectItem>
   267→            <SelectItem value="all">All statuses</SelectItem>
   268→          </SelectContent>
   269→        </Select>
   270→        <div className="flex flex-wrap items-center gap-2">
   271→          <Chip label="Birthdays ≤14d" icon={Gift} active={birthdayChip} onClick={() => setBirthdayChip(!birthdayChip)} />
   272→          <Chip label="Inactive 45d+" icon={Hourglass} active={inactiveChip} onClick={() => setInactiveChip(!inactiveChip)} />
   273→        </div>
   274→      </div>
   275→
   276→      {/* Main area: table + segments panel */}
   277→      <div className="flex items-start gap-4">
   278→        <div className="min-w-0 flex-1 space-y-2">
   279→          {listError && (
   280→            <Alert variant="destructive">
   281→              <AlertTitle>Could not load customers</AlertTitle>
   282→              <AlertDescription className="flex flex-wrap items-center gap-3">
   283→                <span className="flex-1">{listError}</span>
   284→                <Button size="sm" variant="outline" onClick={() => void fetchPage(null, "replace")}>Retry</Button>
   285→              </AlertDescription>
   286→            </Alert>
   287→          )}
   288→
   289→          <div className="overflow-x-auto rounded-xl border bg-card">
   290→            <Table>
   291→              <TableHeader>
   292→                <TableRow className="hover:bg-transparent">
   293→                  <TableHead>Customer</TableHead>
   294→                  <TableHead className="hidden md:table-cell">Tags</TableHead>
   295→                  <TableHead className="text-right">Visits</TableHead>
   296→                  <TableHead className="text-right">Lifetime value</TableHead>
   297→                  <TableHead className="hidden text-right sm:table-cell">Points</TableHead>
   298→                  <TableHead className="hidden lg:table-cell">Last visit</TableHead>
   299→                  <TableHead className="hidden md:table-cell">Source</TableHead>
   300→                </TableRow>
   301→              </TableHeader>
   302→              <TableBody>
   303→                {loading ? (
   304→                  Array.from({ length: 6 }).map((_, i) => (
   305→                    <TableRow key={`sk-${i}`}><TableCell colSpan={7}><Skeleton className="h-9 w-full" /></TableCell></TableRow>
   306→                  ))
   307→                ) : items.length ? (
   308→                  items.map((c) => <CustomerRow key={c.id} customer={c} onOpen={() => setDetailId(c.id)} />)
   309→                ) : (
   310→                  <TableRow className="hover:bg-transparent">
   311→                    <TableCell colSpan={7} className="h-32 text-center text-sm text-muted-foreground">No customers match the current filters.</TableCell>
   312→                  </TableRow>
   313→                )}
   314→              </TableBody>
   315→            </Table>
   316→          </div>
   317→
   318→          <div className="flex flex-wrap items-center justify-between gap-2">
   319→            <p className="text-[11px] text-muted-foreground">Cursor-paginated — no offset scans · showing {items.length}{hasMore ? "+" : ""} customers</p>
   320→            {hasMore && (
   321→              <Button variant="outline" className="h-10" disabled={loadingMore} onClick={loadMore}>
   322→                {loadingMore ? "Loading…" : `Load ${PAGE_SIZE} more`}
   323→              </Button>
   324→            )}
   325→          </div>
   326→        </div>
   327→
   328→        <SegmentsPanel
   329→          segments={segments} loading={segmentsLoading} activeSegmentId={segmentId}
   330→          onToggle={(id) => setSegmentId(segmentId === id ? "" : id)}
   331→          onDelete={deleteSegment}
   332→          onNew={() => setSegmentOpen(true)}

   336→      <AddCustomerDialog open={addOpen} onOpenChange={setAddOpen} onCreated={() => void fetchPage(null, "replace")} />

   338→      <CustomerDrawer customerId={detailId} onClose={() => setDetailId(null)} onChanged={refreshSilently} />

   383→function SegmentsPanel({ segments, loading, activeSegmentId, onToggle, onDelete, onNew }: {
   384→  segments: SegmentDto[]; loading: boolean; activeSegmentId: string;
   385→  onToggle: (id: string) => void; onDelete: (segment: SegmentDto) => void; onNew: () => void;
   386→}) {

   389→      <CardHeader className="pb-3">
   390→        <CardTitle className="flex items-center justify-between text-base">

   395→      </CardHeader>
   396→      <CardContent className="max-h-[420px] overflow-y-auto">
   397→        {loading ? (
   398→          <div className="space-y-2">{[0, 1, 2].map((i) => <Skeleton key={i} className="h-16 w-full" />)}</div>
   399→        ) : segments.length ? (
   400→          <div className="space-y-2">
   401→            {segments.map((s) => (
   402→              <div key={s.id} className={`rounded-lg border p-3 ${activeSegmentId === s.id ? "border-[var(--brand)] bg-[var(--brand)]/5" : ""}`}>
   403→                <button type="button" className="flex w-full items-center justify-between gap-2 text-left" onClick={() => onToggle(s.id)}>
   404→                  <span className="text-sm font-medium">{s.name}</span>
   405→                  <Badge variant={activeSegmentId === s.id ? "default" : "secondary"}
   406→                    className={`text-[10px] tabular-nums ${activeSegmentId === s.id ? "border-transparent bg-[var(--brand)] text-white" : ""}`}>
   407→                    {s.memberCount}
   408→                  </Badge>
   409→                </button>
   410→                {s.description && <p className="mt-1 line-clamp-2 text-xs text-muted-foreground">{s.description}</p>}
   411→                <div className="mt-1.5 space-y-0.5">
   412→                  {s.rules.map((r, i) => (
   413→                    <p key={i} className="font-mono text-[10px] text-muted-foreground">{r.attr} {r.op} {ruleValueText(r.value)}</p>

   415→                </div>
   416→                <Button variant="ghost" size="sm" className="mt-1.5 h-7 gap-1 px-1.5 text-xs text-muted-foreground hover:text-destructive" onClick={() => onDelete(s)}>
   417→                  <Trash2 className="h-3.5 w-3.5" /> Delete segment
   418→                </Button>
   419→              </div>
   420→            ))}
   421→          </div>
   422→        ) : (

   424→        )}
   425→      </CardContent>
   426→    </Card>
   427→  );

   794→            </Dialog>
   795→          </>
   796→        ) : null}
   797→      </SheetContent>
   798→    </Sheet>

   802→function TimelineTab({ customerId }: { customerId: string }) {
   803→  const [entries, setEntries] = useState<TimelineEntry[] | null>(null);
   804→  const [error, setError] = useState<string | null>(null);
   805→
   806→  useEffect(() => {
   807→    let active = true;
   808→    setEntries(null);
   809→    setError(null);
   810→    apiGet<{ timeline: TimelineEntry[] }>(`/api/customers/${customerId}/timeline`)
   811→      .then((res) => { if (active) setEntries(res.timeline); })
   812→      .catch((err: Error) => { if (active) setError(err.message); });
   813→    return () => { active = false; };
   814→  }, [customerId]);
   815→
   816→  if (error) {
   817→    return (
   818→      <Alert variant="destructive">
   819→        <AlertTitle>Could not load timeline</AlertTitle>
   820→        <AlertDescription>{error}</AlertDescription>
   821→      </Alert>
   822→    );
   823→  }
   824→  if (!entries) {
   825→    return <div className="space-y-2">{[0, 1, 2, 3, 4].map((i) => <Skeleton key={i} className="h-14 w-full" />)}</div>;
   826→  }
   827→  if (!entries.length) {
   828→    return <p className="py-10 text-center text-sm text-muted-foreground">Nothing on the timeline yet — record a visit or log feedback.</p>;
   829→  }
   830→  return (
   831→    <div className="max-h-[480px] space-y-3 overflow-y-auto pr-1">
   832→      {entries.map((e, i) => (
   833→        <div key={i} className={`rounded-r-md border-l-2 pl-3 ${toneBorder(e.tone)}`}>
   834→          <div className="flex items-center justify-between gap-2">
   835→            <span className="font-mono text-[10px] uppercase tracking-wide text-muted-foreground">{e.type}</span>

   999→type RuleDraft = { attr: string; op: string; value: string };
  1000→
  1001→function SegmentBuilderDialog({ open, onOpenChange, onCreated }: { open: boolean; onOpenChange: (o: boolean) => void; onCreated: () => void }) {
  1002→  const { toast } = useToast();
  1003→  const [name, setName] = useState("");
  1004→  const [description, setDescription] = useState("");
  1005→  const [rules, setRules] = useState<RuleDraft[]>([{ attr: "visitCount", op: "gte", value: "1" }]);
  1006→  const [tags, setTags] = useState<TagDto[]>([]);
  1007→  const [saving, setSaving] = useState(false);
  1008→
  1009→  useEffect(() => {
  1010→    if (!open) return;
  1011→    setName("");
  1012→    setDescription("");
  1013→    setRules([{ attr: "visitCount", op: "gte", value: "1" }]);
  1014→    apiGet<{ tags: TagDto[] }>("/api/tags").then((res) => setTags(res.tags)).catch(() => setTags([]));
  1015→  }, [open]);
  1016→
  1017→  // point tag rules at a real tag once options are known
  1018→  useEffect(() => {
  1019→    setRules((prev) => prev.map((r) => (r.attr === "tag" && !r.value && tags.length ? { ...r, value: tags[0].id } : r)));
  1020→  }, [tags]);
  1021→
  1022→  const updateRule = (index: number, patch: Partial<RuleDraft>) => {
  1023→    setRules((prev) =>
  1024→      prev.map((r, i) => {
  1025→        if (i !== index) return r;
  1026→        const next = { ...r, ...patch };
  1027→        if (patch.attr !== undefined && patch.attr !== r.attr) {
  1028→          next.value = patch.attr === "tag" ? (tags[0]?.id ?? "") : "";
  1029→          if (["source", "status", "tag"].includes(next.attr) && !["eq", "neq"].includes(next.op)) next.op = "eq";
  1030→        }
  1031→        return next;
  1032→      })
  1033→    );
  1034→  };
  1035→
  1036→  const removeRule = (index: number) => setRules((prev) => (prev.length > 1 ? prev.filter((_, i) => i !== index) : prev));
  1037→
  1038→  const submit = async () => {
  1039→    if (!name.trim()) {
  1040→      toast({ title: "Segment name is required", variant: "destructive" });
  1041→      return;
  1042→    }
  1043→    const payloadRules: SegmentRuleDto[] = [];
  1044→    for (const r of rules) {
  1045→      const raw = r.value.trim();
  1046→      if (!raw) {
  1047→        toast({ title: "Missing rule value", description: `Set a value for the ${r.attr} rule`, variant: "destructive" });
  1048→        return;
  1049→      }
  1050→      if (r.attr === "tag") {
  1051→        payloadRules.push({ attr: r.attr, op: r.op, value: [raw] });
  1052→      } else if (r.op === "between") {
  1053→        const parts = raw.split(",").map((p) => p.trim()).filter(Boolean);
  1054→        if (parts.length !== 2) {
  1055→          toast({ title: "Between needs two bounds", description: 'Use the "min,max" format, e.g. 20,60', variant: "destructive" });
  1056→          return;
  1057→        }
  1058→        payloadRules.push({ attr: r.attr, op: r.op, value: parts });
  1059→      } else if (NUMERIC_ATTRS.includes(r.attr)) {
  1060→        const n = Number(raw);
  1061→        if (!Number.isFinite(n)) {
  1062→          toast({ title: "Numeric rule", description: `${r.attr} needs a number`, variant: "destructive" });
  1063→          return;
  1064→        }
  1065→        payloadRules.push({ attr: r.attr, op: r.op, value: n });
  1066→      } else {
  1067→        payloadRules.push({ attr: r.attr, op: r.op, value: raw });
  1068→      }
  1069→    }
  1070→    setSaving(true);
  1071→    try {
  1072→      await apiPost("/api/segments", { name: name.trim(), description: description.trim() || null, rules: payloadRules });
  1073→      toast({ title: "Segment created", description: "Membership is recomputed live by the segmentation engine" });
  1074→      onOpenChange(false);
  1075→      onCreated();
  1076→    } catch (err) {
  1077→      toast({ title: "Could not create segment", description: (err as Error).message, variant: "destructive" });
  1078→    } finally {
  1079→      setSaving(false);
  1080→    }
  1081→  };
  1082→
  1083→  return (
  1084→    <Dialog open={open} onOpenChange={onOpenChange}>
  1085→      <DialogContent className="sm:max-w-xl">
  1086→        <DialogHeader>
  1087→          <DialogTitle>New segment</DialogTitle>
  1088→          <DialogDescription>Rules run against live customer attributes — no membership rows are stored.</DialogDescription>
  1089→        </DialogHeader>
  1090→        <div className="space-y-3">
  1091→          <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
  1092→            <div className="space-y-1.5">
  1093→              <Label htmlFor="seg-name">Name</Label>
  1094→              <Input id="seg-name" className="h-10" placeholder="e.g. VIP regulars" value={name} onChange={(e) => setName(e.target.value)} />
  1095→            </div>
  1096→            <div className="space-y-1.5">
  1097→              <Label htmlFor="seg-desc">Description</Label>
  1098→              <Input id="seg-desc" className="h-10" placeholder="Optional, shown in the segments panel" value={description} onChange={(e) => setDescription(e.target.value)} />
  1099→            </div>
  1100→          </div>
  1101→          <div className="space-y-2">
  1102→            <Label>Rules (all must match)</Label>
  1103→            {rules.map((r, i) => (
  1104→              <div key={i} className="grid grid-cols-[1fr_1fr] items-center gap-2 sm:grid-cols-[9rem_8rem_1fr_2.75rem]">
  1105→                <Select value={r.attr} onValueChange={(v) => updateRule(i, { attr: v })}>
  1106→                  <SelectTrigger aria-label="Rule attribute" className="h-10 w-full"><SelectValue /></SelectTrigger>
  1107→                  <SelectContent>
  1108→                    {ATTR_OPTIONS.map((a) => <SelectItem key={a.value} value={a.value}>{a.label}</SelectItem>)}
  1109→                  </SelectContent>
  1110→                </Select>
  1111→                <Select value={r.op} onValueChange={(v) => updateRule(i, { op: v })}>
  1112→                  <SelectTrigger aria-label="Rule operator" className="h-10 w-full"><SelectValue /></SelectTrigger>
  1113→                  <SelectContent>
  1114→                    {OP_OPTIONS.map((o) => <SelectItem key={o.value} value={o.value}>{o.label}</SelectItem>)}
  1115→                  </SelectContent>
  1116→                </Select>
  1117→                {r.attr === "tag" ? (
  1118→                  <Select value={r.value} onValueChange={(v) => updateRule(i, { value: v })}>
  1119→                    <SelectTrigger aria-label="Tag for the rule" className="col-span-2 h-10 w-full sm:col-span-1"><SelectValue placeholder="Pick a tag" /></SelectTrigger>
  1120→                    <SelectContent>
  1121→                      {tags.length ? (
  1122→                        tags.map((t) => <SelectItem key={t.id} value={t.id}>{t.name}</SelectItem>)
  1123→                      ) : (
  1124→                        <SelectItem value="none" disabled>No tags yet</SelectItem>
  1125→                      )}
  1126→                    </SelectContent>
  1127→                  </Select>
  1128→                ) : (
  1129→                  <Input
  1130→                    aria-label="Rule value" className="col-span-2 h-10 sm:col-span-1"
  1131→                    placeholder={r.op === "between" ? "min,max — e.g. 20,60" : NUMERIC_ATTRS.includes(r.attr) ? "number" : "value"}
  1132→                    value={r.value} onChange={(e) => updateRule(i, { value: e.target.value })}
  1133→                  />
  1134→                )}
  1135→                <Button type="button" variant="ghost" size="icon" className="col-span-2 h-10 w-full sm:col-span-1 sm:w-11"
  1136→                  disabled={rules.length === 1} onClick={() => removeRule(i)} aria-label="Remove rule">
  1137→                  <X className="h-4 w-4" />
  1138→                </Button>
  1139→              </div>
  1140→            ))}
  1141→            <Button type="button" variant="outline" className="h-10 w-full gap-1.5" onClick={() => setRules((prev) => [...prev, { attr: "visitCount", op: "gte", value: "1" }])}>
  1142→              <Plus className="h-4 w-4" /> Add rule
  1143→            </Button>
  1144→          </div>
  1145→        </div>
  1146→        <DialogFooter>
  1147→          <Button variant="outline" className="h-10" onClick={() => onOpenChange(false)}>Cancel</Button>
  1148→          <Button className="h-10 bg-[var(--brand)] text-white hover:opacity-90" disabled={saving} onClick={() => void submit()}>
  1149→            {saving ? "Creating…" : "Create segment"}
  1150→          </Button>

  1153→    </Dialog>
